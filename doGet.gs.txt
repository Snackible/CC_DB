// Map the ?sheet= value the dashboard sends to the spreadsheet that holds
// it. Add an entry here for every sheet the frontend needs.
const SPREADSHEET_MAP = {
  delhivery: '18-6td6ziN3RhcLi5W88yGiVXQbyWzRqi83Eqp8EflJ8'
};
const CACHE_SECONDS = 120; // how long a sheet read is reused before re-reading live

function doGet(e) {
  const sheetParam = (e.parameter.sheet || '').toLowerCase();
  const callback = e.parameter.callback; // present when the frontend loads via JSONP

  const data = getCached(sheetParam, () => readSheet(sheetParam));
  return jsonpResponse(data, callback);
}

function readSheet(sheetParam) {
  const spreadsheetId = SPREADSHEET_MAP[sheetParam];
  if (!spreadsheetId) {
    return { error: 'Unknown sheet: ' + sheetParam, rows: [] };
  }

  // First tab in the spreadsheet (gid=0), regardless of its name.
  const sheet = SpreadsheetApp.openById(spreadsheetId).getSheets()[0];
  if (!sheet) {
    return { error: 'No tabs found in spreadsheet: ' + spreadsheetId, rows: [] };
  }

  const values = sheet.getDataRange().getValues();
  const headers = values[0].map(h => toCamelCase(String(h)));
  const rows = values.slice(1).map(row => {
    const obj = {};
    headers.forEach((key, i) => { obj[key] = row[i]; });
    return obj;
  });

  return { sheet: sheetParam, count: rows.length, rows: rows };
}

// Reuses the last read for CACHE_SECONDS instead of hitting the spreadsheet
// on every request — Apps Script Web Apps are slow enough under load that
// this matters a lot for repeat/near-simultaneous requests.
function getCached(key, compute) {
  const cache = CacheService.getScriptCache();
  const cached = cache.get(key);
  if (cached) return JSON.parse(cached);

  const data = compute();
  cache.put(key, JSON.stringify(data), CACHE_SECONDS);
  return data;
}

// "Order ID" -> "orderId", "CC Remark" -> "ccRemark"
function toCamelCase(header) {
  return header
    .trim()
    .replace(/[^a-zA-Z0-9]+(.)/g, (_, c) => c.toUpperCase())
    .replace(/^[A-Z]/, c => c.toLowerCase());
}

// Wraps the JSON as `callback(...)` and serves it as JavaScript when the
// frontend loads this via a <script> tag (JSONP) instead of fetch() — plain
// JSON is still returned for direct navigation/testing when no callback is given.
function jsonpResponse(obj, callback) {
  if (callback) {
    return ContentService
      .createTextOutput(callback + '(' + JSON.stringify(obj) + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
